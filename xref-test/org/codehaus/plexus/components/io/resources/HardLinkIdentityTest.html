16  package org.codehaus.plexus.components.io.resources;
17  
18  import java.io.ByteArrayInputStream;
19  import java.nio.file.Files;
20  import java.nio.file.Path;
21  
22  import org.codehaus.plexus.components.io.functions.ContentSupplier;
23  import org.codehaus.plexus.components.io.functions.HardLinkIdentitySupplier;
24  import org.codehaus.plexus.components.io.functions.NameSupplier;
25  import org.codehaus.plexus.components.io.resources.proxy.ProxyFactory;
26  import org.junit.jupiter.api.Test;
27  import org.junit.jupiter.api.io.TempDir;
28  
29  import static org.junit.jupiter.api.Assertions.*;
30  import static org.junit.jupiter.api.Assumptions.assumeTrue;
31  
32  class HardLinkIdentityTest {
33      @TempDir
34      Path temp;
35  
36      /** Checks filesystem identity and invalidation when effective content changes. */
37      @Test
38      void identifiesOnlyOriginalContent() throws Exception {
39          Path first = Files.write(temp.resolve("first"), new byte[] {1});
40          Path second = temp.resolve("second");
41          try {
42              Files.createLink(second, first);
43          } catch (java.io.IOException | UnsupportedOperationException e) {
44              assumeTrue(false, "Hard links unavailable: " + e);
45          }
46          PlexusIoResource a = ResourceFactory.createResource(first.toFile());
47          PlexusIoResource b = ResourceFactory.createResource(second.toFile());
48          Object identity = ((HardLinkIdentitySupplier) a).getHardLinkIdentity();
49          assumeTrue(identity != null, "Filesystem does not expose file keys");
50          assertEquals(identity, ((HardLinkIdentitySupplier) b).getHardLinkIdentity());
51          PlexusIoResource transformed = ResourceFactory.createResource(
52                  first.toFile(), (resource, input) -> new ByteArrayInputStream(new byte[] {2}));
53          assertNull(((HardLinkIdentitySupplier) transformed).getHardLinkIdentity());
54          PlexusIoResource supplied = ResourceFactory.createResource(
55                  first.toFile(),
56                  "custom",
57                  () -> new ByteArrayInputStream(new byte[] {3}),
58                  (org.codehaus.plexus.components.io.functions.InputStreamTransformer) null);
59          assertNull(((HardLinkIdentitySupplier) supplied).getHardLinkIdentity());
60      }
61  
62      /** Transparent renaming keeps identity while a replacement stream invalidates it. */
63      @Test
64      void proxiesRespectContentOverrides() throws Exception {
65          Path file = Files.write(temp.resolve("file"), new byte[] {1});
66          PlexusIoResource original = ResourceFactory.createResource(file.toFile());
67          Object identity = ((HardLinkIdentitySupplier) original).getHardLinkIdentity();
68          PlexusIoResource renamed = ProxyFactory.createProxy(original, (NameSupplier) () -> "renamed");
69          assertEquals(identity, ((HardLinkIdentitySupplier) renamed).getHardLinkIdentity());
70          PlexusIoResource replaced =
71                  ProxyFactory.createProxy(original, (ContentSupplier) () -> new ByteArrayInputStream(new byte[] {2}));
72          assertNull(((HardLinkIdentitySupplier) replaced).getHardLinkIdentity());
73      }
74  
75      /** Collection resolution must explicitly preserve or discard the source guarantee. */
76      @Test
77      void deferredResourcesRespectTransformers() throws Exception {
78          Path file = Files.write(temp.resolve("file"), new byte[] {1});
79          PlexusIoResource original = ResourceFactory.createResource(file.toFile());
80          Object identity = ((HardLinkIdentitySupplier) original).getHardLinkIdentity();
81          PlexusIoResourceCollection owner = new PlexusIoFileResourceCollection();
82          assertEquals(
83                  identity,
84                  ((HardLinkIdentitySupplier) new Deferred(original, owner, false).asResource()).getHardLinkIdentity());
85          assertNull(((HardLinkIdentitySupplier) new Deferred(original, owner, true).asResource()).getHardLinkIdentity());
86      }
87  }
